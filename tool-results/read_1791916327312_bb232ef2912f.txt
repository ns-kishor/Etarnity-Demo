    20→async function registerMedia(
    21→  fileName: string,
    22→  altText: string
    23→): Promise<number | null> {
    24→  const full = path.join(UPLOAD_DIR, fileName);
    25→  if (!existsSync(full)) {
    26→    console.warn(`  ! missing media file: ${fileName} (skipped)`);
    27→    return null;
    28→  }
    29→  const stat = statSync(full);
    30→  const row = await db.mediaItem.create({
    31→    data: {
    32→      storageKey: fileName,
    33→      fileName,
    34→      mimeType: "image/png",
    35→      size: stat.size,
    36→      altText,
    37→    },
    38→  });
    39→  return row.id;
    40→}
